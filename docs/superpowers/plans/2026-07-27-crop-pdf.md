# Crop PDF Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a new "Crop PDF" tool — a single drag/resize crop box (drawn on page 1's live preview) applied to every page of the uploaded PDF via `pdf-lib`'s `CropBox`, with no rasterization and no worker protocol change.

**Architecture:** New engine file (`engine/cropPdf.ts`, synchronous pdf-lib, mirrors `engine/flattenPdf.ts`) + new shell component (`CropShell.tsx`, `CompressShell`'s upload→options→processing→done phase machine, with the options phase porting `SignShell`'s exact draggable/resizable fractional-box interaction) + two Astro pages + i18n entries + a home-page grid card.

**Tech Stack:** React (TSX), TypeScript, `pdf-lib` (already a dependency), existing `JobController.previewPage` for the drag-target preview image (no new worker message).

## Global Constraints

- One crop box, applied identically (as a fraction of each page's own width/height) to every page — no per-page crop, no preset ratios, no undo/redo (spec: `docs/superpowers/specs/2026-07-27-crop-pdf-design.md`, confirmed out of scope).
- `MediaBox` must stay untouched — only `CropBox` is set (non-destructive, reversible by any other PDF tool).
- No new `JobController` method, no new `UiToWorkerMessage`/`WorkerToUiMessage` variant. The only worker interaction is the existing `previewPage(file, page, dpi?)` call, identical to how `SignShell` already uses it.
- New `ToolCopy` entry (`cropCopy`) must include `faqTitle` and `faq` (some older entries in `toolCopy.ts` are missing these — do not repeat that gap here).
- Tagline text must render on one line at the shared 720px tool-page container — keep it under ~85 characters (the safe budget measured this session across every other tool's tagline).
- Verification is `tsc --noEmit` + `npm run build` + `npm test` (vitest) — per `CLAUDE.md`, do not use Playwright/browser automation for visual review; manual check is called out explicitly where needed.

Create `web/src/engine/cropPdf.ts`:

```ts
import { PDFDocument } from 'pdf-lib';

export interface CropBoxFrac {
  xFrac: number;
  yFrac: number;
  widthFrac: number;
  heightFrac: number;
}

export interface CropPdfResult {
  output: Blob;
  outputName: string;
  durationMs: number;
  pageCount: number;
}

export async function cropPdf(file: File, box: CropBoxFrac): Promise<CropPdfResult> {
  const start = performance.now();

  const arrayBuffer = await file.arrayBuffer();
  const doc = await PDFDocument.load(new Uint8Array(arrayBuffer), { ignoreEncryption: true });

  if (doc.isEncrypted) {
    throw new Error('ENCRYPTED_PDF_UNSUPPORTED');
  }

  const pages = doc.getPages();
  if (pages.length === 0) {
    throw new Error('ZERO_PAGE_PDF');
  }

  for (const page of pages) {
    const { width: pageWidth, height: pageHeight } = page.getSize();

    // Screen-space (top-down) fraction -> PDF's bottom-up coordinate space.
    const pdfX = box.xFrac * pageWidth;
    const pdfY = pageHeight - (box.yFrac + box.heightFrac) * pageHeight;
    const pdfWidth = box.widthFrac * pageWidth;
    const pdfHeight = box.heightFrac * pageHeight;

    // MediaBox is left untouched on purpose — CropBox is the non-destructive,
    // viewer-respected visible-area override (spec: 2026-07-27-crop-pdf-design.md).
    page.setCropBox(pdfX, pdfY, pdfWidth, pdfHeight);
  }

  const savedBytes = await doc.save();
  const output = new Blob([savedBytes], { type: 'application/pdf' });
  const baseName = (file.name || 'document.pdf').replace(/\.[^/.]+$/, '');
  const outputName = `${baseName}_cropped.pdf`;
  const durationMs = Math.round(performance.now() - start);

  return {
    output,
    outputName,
    durationMs,
    pageCount: pages.length,
  };
}
```

```tsx
import { useCallback, useEffect, useRef, useState } from 'react';
import { JobController } from '../app/JobController';
import { DropZone } from './DropZone';
import { PrivacyLine } from './PrivacyLine';
import { Button } from './ui/Button';
import { Toast, type ToastData } from './Toast';
import { triggerDownload } from '../app/download';
import { validatePdfFile } from '../app/validators';
import { cropPdf, type CropBoxFrac } from '../engine/cropPdf';
import type { Strings } from '../i18n/en';
import { en } from '../i18n/en';
import { Crop as CropIcon, Check, Download, RefreshCw } from 'lucide-react';

type Phase = 'upload' | 'options' | 'processing' | 'done';

interface Props {
  t?: Strings;
}

const DEFAULT_BOX: CropBoxFrac = {
  xFrac: 0.1,
  yFrac: 0.1,
  widthFrac: 0.8,
  heightFrac: 0.8,
};

export function CropShell({ t = en }: Props) {
  const [phase, setPhase] = useState<Phase>('upload');
  const [file, setFile] = useState<File | null>(null);
  const [toast, setToast] = useState<ToastData | null>(null);
  const [box, setBox] = useState<CropBoxFrac>(DEFAULT_BOX);
  const [previewUrl, setPreviewUrl] = useState<string | null>(null);
  const [isLoadingPreview, setIsLoadingPreview] = useState(false);
  const [isPlacingBox, setIsPlacingBox] = useState(false);
  const [output, setOutput] = useState<{ blob: Blob; name: string; pageCount: number } | null>(null);

  const previewContainerRef = useRef<HTMLDivElement | null>(null);
  // Same guard as SignShell: a drag/resize release always fires a native
  // click on mouseup right after — without this, the box re-centers on
  // the release point instead of staying where it was dropped.
  const suppressNextClickRef = useRef(false);

  const controllerRef = useRef<JobController | null>(null);
  const getController = useCallback((): JobController => {
    if (!controllerRef.current) {
      controllerRef.current = new JobController({
        onFatal: () => setToast({ kind: 'error', message: t.corruptFile }),
        onUnavailable: () => setToast({ kind: 'error', message: t.toolUnavailable }),
      });
    }
    return controllerRef.current;
  }, [t]);

  useEffect(() => {
    return () => controllerRef.current?.dispose();
  }, []);

  useEffect(() => {
    if (!file) {
      setPreviewUrl(null);
      return;
    }
    let active = true;
    setIsLoadingPreview(true);
    getController()
      .previewPage(file, 1, 400)
      .then((blob) => {
        if (active) {
          setPreviewUrl(URL.createObjectURL(blob));
          setIsLoadingPreview(false);
        }
      })
      .catch(() => {
        if (active) setIsLoadingPreview(false);
      });
    return () => { active = false; };
  }, [file, getController]);

  const handleBoxMouseDown = (e: React.MouseEvent) => {
    e.stopPropagation();
    e.preventDefault();
    const container = previewContainerRef.current;
    if (!container) return;

    const startX = e.clientX;
    const startY = e.clientY;
    const startXFrac = box.xFrac;
    const startYFrac = box.yFrac;
    const rect = container.getBoundingClientRect();
    setIsPlacingBox(true);

    const onMouseMove = (moveEvent: MouseEvent) => {
      const deltaX = moveEvent.clientX - startX;
      const deltaY = moveEvent.clientY - startY;
      const newXFrac = Math.max(0, Math.min(1 - box.widthFrac, startXFrac + deltaX / rect.width));
      const newYFrac = Math.max(0, Math.min(1 - box.heightFrac, startYFrac + deltaY / rect.height));
      setBox((prev) => ({ ...prev, xFrac: newXFrac, yFrac: newYFrac }));
    };

    const onMouseUp = () => {
      setIsPlacingBox(false);
      suppressNextClickRef.current = true;
      window.removeEventListener('mousemove', onMouseMove);
      window.removeEventListener('mouseup', onMouseUp);
    };

    window.addEventListener('mousemove', onMouseMove);
    window.addEventListener('mouseup', onMouseUp);
  };

  const handleResizeMouseDown = (e: React.MouseEvent) => {
    e.stopPropagation();
    e.preventDefault();
    const container = previewContainerRef.current;
    if (!container) return;

    const startX = e.clientX;
    const startY = e.clientY;
    const startWFrac = box.widthFrac;
    const startHFrac = box.heightFrac;
    const rect = container.getBoundingClientRect();
    setIsPlacingBox(true);

    const onMouseMove = (moveEvent: MouseEvent) => {
      const deltaX = moveEvent.clientX - startX;
      const deltaY = moveEvent.clientY - startY;
      const newWFrac = Math.max(0.05, Math.min(1 - box.xFrac, startWFrac + deltaX / rect.width));
      const newHFrac = Math.max(0.05, Math.min(1 - box.yFrac, startHFrac + deltaY / rect.height));
      setBox((prev) => ({ ...prev, widthFrac: newWFrac, heightFrac: newHFrac }));
    };

    const onMouseUp = () => {
      setIsPlacingBox(false);
      suppressNextClickRef.current = true;
      window.removeEventListener('mousemove', onMouseMove);
      window.removeEventListener('mouseup', onMouseUp);
    };

    window.addEventListener('mousemove', onMouseMove);
    window.addEventListener('mouseup', onMouseUp);
  };

  const handlePreviewClick = (e: React.MouseEvent<HTMLDivElement>) => {
    if (suppressNextClickRef.current) {
      suppressNextClickRef.current = false;
      return;
    }
    const rect = e.currentTarget.getBoundingClientRect();
    const clickXFrac = (e.clientX - rect.left) / rect.width;
    const clickYFrac = (e.clientY - rect.top) / rect.height;
    const xFrac = Math.max(0, Math.min(1 - box.widthFrac, clickXFrac - box.widthFrac / 2));
    const yFrac = Math.max(0, Math.min(1 - box.heightFrac, clickYFrac - box.heightFrac / 2));
    setBox((prev) => ({ ...prev, xFrac, yFrac }));
  };

  const addFile = useCallback(async (incoming: File[]) => {
    if (incoming.length === 0) return;
    const f = incoming[0];
    const rejection = await validatePdfFile(f);
    if (rejection) {
      setToast({ kind: 'error', message: rejection === 'empty-file' ? t.emptyFile : t.notPdf });
      return;
    }
    setFile(f);
    setBox(DEFAULT_BOX);
    setOutput(null);
    setPhase('options');
  }, [t]);

  const handleCrop = async () => {
    if (!file) return;
    setPhase('processing');
    try {
      const res = await cropPdf(file, box);
      setOutput({ blob: res.output, name: res.outputName, pageCount: res.pageCount });
      setPhase('done');
    } catch (err: any) {
      setToast({
        kind: 'error',
        message: err?.message === 'ENCRYPTED_PDF_UNSUPPORTED' ? t.encryptedFile : t.corruptFile,
      });
      setPhase('options');
    }
  };

  const reset = useCallback(() => {
    setFile(null);
    setOutput(null);
    setPhase('upload');
  }, []);

  return (
    <div className="flex flex-col gap-5">
      {phase === 'upload' && (
        <>
          <DropZone t={t} hasFiles={false} onFiles={addFile} multiple={false} />
          <PrivacyLine t={t} />
        </>
      )}

      {phase === 'options' && file && (
        <div className="phase-enter flex flex-col gap-4">
          <div className="flex items-center gap-3 rounded-2xl border border-amber/30 bg-surface p-4 shadow-[0_0_15px_rgba(232,182,95,0.15)] dark:border-amber-dark/30 dark:bg-surface-dark dark:shadow-[0_0_15px_rgba(232,182,95,0.25)]">
            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-amber/10 text-amber dark:bg-amber-dark/20 dark:text-amber-dark">
              <CropIcon className="h-5 w-5" />
            </div>
            <div className="flex flex-col overflow-hidden min-w-0 flex-1">
              <div className="overflow-x-auto whitespace-nowrap scrollbar-thin text-sm font-medium pr-2" title={file.name}>{file.name}</div>
              <span className="text-xs text-ink-muted dark:text-ink-muted-dark">
                {(file.size / 1024 / 1024).toFixed(2)} MB
              </span>
            </div>
          </div>

          <div className="flex flex-col gap-3 rounded-2xl border bg-surface p-4 dark:bg-surface-dark">
            <span className="text-xs font-semibold uppercase tracking-wide text-ink-muted dark:text-ink-muted-dark">
              {t.lang === 'tr' ? 'Kırpma Alanını Seçin' : 'Select Crop Area'}
            </span>
            <div className="relative flex justify-center bg-surface-2 dark:bg-surface-2-dark rounded border border-ink-muted/20 dark:border-ink-muted-dark/20 p-4 overflow-hidden min-h-[360px]">
              {isLoadingPreview && (
                <div className="absolute inset-0 z-20 flex items-center justify-center bg-black/20 text-xs text-white font-medium">
                  {t.lang === 'tr' ? 'Sayfa önizlemesi yükleniyor...' : 'Loading page preview...'}
                </div>
              )}
              {previewUrl ? (
                <div
                  ref={previewContainerRef}
                  className="relative cursor-crosshair shadow-lg bg-white select-none inline-block max-w-full"
                  onClick={handlePreviewClick}
                >
                  <img src={previewUrl} alt="PDF Page Preview" className="max-h-[500px] w-auto block pointer-events-none" />
                  <div
                    className={`absolute rounded-md flex items-center justify-center cursor-move select-none overflow-hidden transition-all duration-300 ease-[cubic-bezier(0.22,1,0.36,1)] ${
                      isPlacingBox
                        ? 'scale-[1.02] border border-amber bg-amber/10 shadow-[0_0_15px_rgba(232,182,95,0.25)] dark:border-amber-dark dark:bg-amber-dark/15 z-10'
                        : 'border border-amber/40 bg-transparent hover:border-amber/70 dark:border-amber-dark/40 dark:hover:border-amber-dark/70'
                    }`}
                    style={{
                      left: `${box.xFrac * 100}%`,
                      top: `${box.yFrac * 100}%`,
                      width: `${box.widthFrac * 100}%`,
                      height: `${box.heightFrac * 100}%`,
                    }}
                    onMouseDown={handleBoxMouseDown}
                  >
                    <div
                      className="absolute bottom-0 right-0 w-4 h-4 bg-amber dark:bg-amber-dark rounded-tl-sm cursor-se-resize flex items-center justify-center shadow-md hover:scale-110 transition-transform z-10"
                      onMouseDown={handleResizeMouseDown}
                    />
                  </div>
                </div>
              ) : (
                <div className="flex items-center justify-center text-xs text-ink-muted min-h-[250px]">
                  {t.lang === 'tr' ? 'Önizleme hazırlanamadı' : 'Preview not available'}
                </div>
              )}
            </div>
            <span className="text-[11px] text-center text-ink-muted dark:text-ink-muted-dark">
              {t.lang === 'tr'
                ? 'Kutuyu sürükleyip taşıyabilir veya sağ alt köşesinden boyutunu değiştirebilirsiniz. Aynı kırpma tüm sayfalara uygulanır.'
                : 'Drag to move the box or resize it from the bottom-right corner. The same crop applies to every page.'}
            </span>
          </div>

          <div className="flex justify-end mt-2">
            <Button onClick={handleCrop}>
              {t.lang === 'tr' ? 'PDF Kırp' : 'Crop PDF'}
            </Button>
          </div>
        </div>
      )}

      {phase === 'processing' && (
        <div className="phase-enter flex flex-col gap-3">
          <div className="flex items-baseline justify-between text-xs text-ink-muted dark:text-ink-muted-dark">
            <span>{t.converting}</span>
          </div>
          <div className="h-1 overflow-hidden rounded-lg bg-surface border dark:bg-surface-dark">
            <div className="h-full w-full origin-left animate-custom-pulse bg-ink dark:bg-ink-dark" />
          </div>
        </div>
      )}

      {phase === 'done' && output && (
        <div className="phase-enter flex flex-col gap-5 rounded-2xl border border-amber/30 bg-surface p-6 shadow-[0_0_15px_rgba(232,182,95,0.15)] dark:border-amber-dark/30 dark:bg-surface-dark dark:shadow-[0_0_15px_rgba(232,182,95,0.25)]">
          <div className="flex flex-col items-center justify-center text-center gap-4 py-2">
            <div className="flex h-12 w-12 items-center justify-center rounded-full bg-success/10 text-success">
              <Check className="h-6 w-6" />
            </div>
            <div className="flex flex-col gap-1">
              <h3 className="text-lg font-semibold">
                {t.lang === 'tr' ? 'PDF Başarıyla Kırpıldı!' : 'PDF Successfully Cropped!'}
              </h3>
              <p className="text-sm text-ink-muted dark:text-ink-muted-dark">
                {t.lang === 'tr'
                  ? `${output.pageCount} sayfa kırpıldı ve indirmeye hazır.`
                  : `${output.pageCount} page${output.pageCount > 1 ? 's' : ''} cropped and ready to download.`}
              </p>
            </div>
            <div className="flex flex-wrap items-center justify-center gap-3 pt-2">
              <Button variant="ghost" onClick={reset} className="flex items-center gap-2">
                <RefreshCw className="h-4 w-4" />
                {t.lang === 'tr' ? 'Yeni Dosya Seç' : 'Crop Another PDF'}
              </Button>
              <Button variant="primary" onClick={() => triggerDownload(output.blob, output.name)} className="flex items-center gap-2">
                <Download className="h-4 w-4" />
                {t.lang === 'tr' ? 'Kırpılmış PDF\'i İndir' : 'Download Cropped PDF'}
              </Button>
            </div>
          </div>
        </div>
      )}

      <Toast toast={toast} onClear={() => setToast(null)} />
    </div>
  );
}
```
